cmake_minimum_required(VERSION 3.21)
project(CelestialAtlas VERSION 0.7.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ATLAS_BUILD_GUI "Build Qt desktop application" ON)
option(ATLAS_ENABLE_CSPICE "Enable NASA NAIF CSPICE precision ephemerides" OFF)
set(CSPICE_ROOT "" CACHE PATH "Unmodified NASA NAIF CSPICE installation root")
set(ATLAS_SPICE_LSK "" CACHE FILEPATH "NAIF leap-second kernel for integration tests")
set(ATLAS_SPICE_SPK "" CACHE FILEPATH "JPL planetary SPK for integration tests")
include(CTest)
add_library(atlas_astronomy STATIC
 src/astronomy/SolarGeometry.cpp
 src/astronomy/MoonGeometry.cpp
 src/astronomy/EarthMoonSystem.cpp
 src/astronomy/SpiceEphemeris.cpp)
if(ATLAS_ENABLE_CSPICE)
 find_path(CSPICE_INCLUDE_DIR NAMES SpiceUsr.h PATHS "${CSPICE_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
 find_file(CSPICE_LIBRARY NAMES cspice.a cspice.lib PATHS "${CSPICE_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
 target_include_directories(atlas_astronomy PRIVATE "${CSPICE_INCLUDE_DIR}")
 target_link_libraries(atlas_astronomy PUBLIC "${CSPICE_LIBRARY}")
 if(UNIX)
  target_link_libraries(atlas_astronomy PUBLIC m)
 endif()
 target_compile_definitions(atlas_astronomy PUBLIC ATLAS_HAS_CSPICE=1)
else()
 target_compile_definitions(atlas_astronomy PUBLIC ATLAS_HAS_CSPICE=0)
endif()
target_include_directories(atlas_astronomy PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(MSVC)
 target_compile_options(atlas_astronomy PRIVATE /W4)
else()
 target_compile_options(atlas_astronomy PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
 add_executable(atlas_astro_tests tests/SolarGeometryTests.cpp)
 target_link_libraries(atlas_astro_tests PRIVATE atlas_astronomy)
 add_test(NAME solar_geometry COMMAND atlas_astro_tests)
 add_executable(atlas_moon_tests tests/MoonGeometryTests.cpp)
 target_link_libraries(atlas_moon_tests PRIVATE atlas_astronomy)
 add_test(NAME moon_geometry COMMAND atlas_moon_tests)
 add_executable(atlas_earth_moon_tests tests/EarthMoonSystemTests.cpp)
 target_link_libraries(atlas_earth_moon_tests PRIVATE atlas_astronomy)
 add_test(NAME earth_moon_scale COMMAND atlas_earth_moon_tests)
 if(ATLAS_ENABLE_CSPICE)
  add_executable(atlas_spice_kernel_tests tests/SpiceKernelTests.cpp)
  target_link_libraries(atlas_spice_kernel_tests PRIVATE atlas_astronomy)
  if(ATLAS_SPICE_LSK AND ATLAS_SPICE_SPK)
   add_test(NAME nasa_spice_kernels COMMAND atlas_spice_kernel_tests "${ATLAS_SPICE_LSK}" "${ATLAS_SPICE_SPK}")
  endif()
 endif()
endif()
if(ATLAS_BUILD_GUI)
 find_package(Qt6 6.4 QUIET COMPONENTS Core Widgets Gui OpenGL OpenGLWidgets Network)
 if(NOT Qt6_FOUND)
  message(WARNING "Qt6 not found: only the astronomy core will be built.")
 else()
  set(CMAKE_AUTOMOC ON)
  set(CMAKE_AUTORCC ON)
  add_executable(CelestialAtlas src/app/main.cpp src/app/MainWindow.hpp src/app/MainWindow.cpp src/app/TextureDownloader.hpp src/app/TextureDownloader.cpp src/render/EarthViewport.hpp src/render/EarthViewport.cpp assets/shaders.qrc)
  target_link_libraries(CelestialAtlas PRIVATE atlas_astronomy Qt6::Core Qt6::Widgets Qt6::Gui Qt6::OpenGL Qt6::OpenGLWidgets Qt6::Network)
  if(WIN32)
   set_target_properties(CelestialAtlas PROPERTIES WIN32_EXECUTABLE TRUE)
  endif()
 endif()
endif()
